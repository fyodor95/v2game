province_event = {
	id = 24001
	title = "EVTNAME24001"
	desc = "EVTDESC24001"
	trigger = {
		controlled_by_rebels = yes
		NOT = {
			is_core = YAK
		}
		owner = {
			NOT = {
				tag = YAK
			}
			YAK = {
				OR = {
					accepted_culture = yakut
					primary_culture = yakut
				}
			}
		}
		OR = {
			AND = {
				culture = yakut
				owner = {
					YAK = {
						exists = no
					}
					NOT = {
						any_owned_province = {
							is_core = YAK
						}
					}
				}
			}
			AND = {
				any_pop = {
					has_pop_culture = yakut
				}
				any_neighbor_province = {
					is_core = YAK
				}
			}
		}
	}
	mean_time_to_happen = {
		months = 4
	}
	option = {
		name = "EVTOPTA24001"
		add_core = YAK
	}
}
