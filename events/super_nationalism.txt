		}

				controlled_by_rebels = no
			}
		}
	}
	mean_time_to_happen = {
		months = 6
	}
	option = {
		name = "EVTOPTA6001"
		release = ABU
		ABU = {
			any_pop = {
				limit = {
					OR = {
						is_accepted_culture = yes
						is_primary_culture = yes
					}

				militancy = -10
			}
		}
		casus_belli = {
			target = ABU
			type = conquest_for_all
			months = 12
		}
	}
